================================================================================

  (use_declaration
    (path (type_identifier))
    (alias (type_identifier))
    (use_items)))

================================================================================

    (path (type_identifier))

        (use_items (identifier) (identifier)))
      (use_item
        (type_identifier)
        (use_items (use_wildcard)))))
  (use_declaration
    (path (type_identifier))
    (use_items (use_wildcard))))
